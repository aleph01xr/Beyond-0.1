/**
 * BEYOND — Premium Boss Ability SFX v2 (Floors 1–5)
 *
 * A more expressive procedural sound-design pass for all 20 boss abilities.
 * The module is intentionally self-contained and uses the caller's existing
 * AudioContext + SFX destination. It does NOT create an audio context, alter
 * SoundManager routing, or change gameplay timing.
 *
 * Integration:
 *   import { playBossAbilitySfx } from './bossSfxRedesign.ts';
 *
 *   const ctx = this.ensureContext();
 *   const dest = this.getDestination();
 *   if (ctx && dest && playBossAbilitySfx(ctx, dest, boss, ability)) return;
 *
 * Keep the existing Floor 6–10 dispatcher as fallback. Disable old Floor 1–5
 * playback paths after integration to prevent doubled sounds.
 *
 * NOTE: This is procedural synthesis, not recorded Foley. Final quality still
 * depends on in-game mix levels, speaker/headphone playback, and playtesting.
 */

type BossId = 'malakor' | 'vaelor' | 'ignis' | 'malice' | 'archon';

const noiseCache = new WeakMap<AudioContext, AudioBuffer>();

const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));

function noise(ctx: AudioContext): AudioBuffer {
  const cached = noiseCache.get(ctx);
  if (cached) return cached;
  const length = Math.max(1, Math.floor(ctx.sampleRate * 1.2));
  const buffer = ctx.createBuffer(1, length, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  let brown = 0;
  for (let i = 0; i < length; i++) {
    const white = Math.random() * 2 - 1;
    brown = (brown + 0.018 * white) / 1.018;
    data[i] = clamp(brown * 3.5, -1, 1);
  }
  noiseCache.set(ctx, buffer);
  return buffer;
}

function saturation(ctx: AudioContext): WaveShaperNode {
  // Each voice gets its own nonlinear stage; sharing one would intermodulate
  // simultaneous sounds and make dense boss attacks harsh.
  const shaper = ctx.createWaveShaper();
  const curve = new Float32Array(2048);
  for (let i = 0; i < curve.length; i++) {
    const x = (i * 2) / (curve.length - 1) - 1;
    curve[i] = Math.tanh(x * 1.65) / Math.tanh(1.65);
  }
  shaper.curve = curve;
  shaper.oversample = '4x';
  return shaper;
}

function gainEnvelope(g: GainNode, t: number, peak: number, attack: number,
                      hold: number, release: number): void {
  const a = Math.max(0.002, attack);
  const h = Math.max(a + 0.002, hold);
  const end = h + Math.max(0.008, release);
  g.gain.cancelScheduledValues(t);
  g.gain.setValueAtTime(0.0001, t);
  g.gain.exponentialRampToValueAtTime(Math.max(0.0002, peak), t + a);
  if (h > a + 0.003) g.gain.setValueAtTime(Math.max(0.0002, peak), t + h);
  g.gain.exponentialRampToValueAtTime(0.0001, t + end);
}

function voice(ctx: AudioContext, dest: AudioNode, t: number, opts: {
  from: number; to?: number; duration: number; peak: number;
  type?: OscillatorType; attack?: number; hold?: number; release?: number;
  detune?: number; pan?: number; drive?: boolean;
}): void {
  const osc = ctx.createOscillator();
  const amp = ctx.createGain();
  const panner = ctx.createStereoPanner();
  osc.type = opts.type ?? 'sine';
  osc.frequency.setValueAtTime(Math.max(1, opts.from), t);
  if (opts.to !== undefined) {
    osc.frequency.exponentialRampToValueAtTime(Math.max(1, opts.to), t + Math.max(0.006, opts.duration));
  }
  if (opts.detune) osc.detune.setValueAtTime(opts.detune, t);
  panner.pan.setValueAtTime(clamp(opts.pan ?? 0, -1, 1), t);
  gainEnvelope(amp, t, opts.peak, opts.attack ?? 0.004,
    opts.hold ?? Math.max(0.008, opts.duration * 0.72),
    opts.release ?? Math.max(0.02, opts.duration * 0.28));
  osc.connect(amp);
  if (opts.drive) {
    const shaper = saturation(ctx);
    amp.connect(shaper);
    shaper.connect(panner);
  } else {
    amp.connect(panner);
  }
  panner.connect(dest);
  const stopAt = t + (opts.hold ?? opts.duration) + (opts.release ?? 0.08) + 0.03;
  osc.start(t);
  osc.stop(stopAt);
}

function texture(ctx: AudioContext, dest: AudioNode, t: number, opts: {
  duration: number; peak: number; filter: BiquadFilterType;
  from: number; to: number; q?: number; attack?: number;
  release?: number; pan?: number; drive?: boolean;
}): void {
  const src = ctx.createBufferSource();
  const filter = ctx.createBiquadFilter();
  const amp = ctx.createGain();
  const panner = ctx.createStereoPanner();
  src.buffer = noise(ctx);
  filter.type = opts.filter;
  filter.Q.setValueAtTime(opts.q ?? 0.8, t);
  filter.frequency.setValueAtTime(Math.max(25, opts.from), t);
  filter.frequency.exponentialRampToValueAtTime(Math.max(25, opts.to), t + opts.duration);
  panner.pan.setValueAtTime(clamp(opts.pan ?? 0, -1, 1), t);
  gainEnvelope(amp, t, opts.peak, opts.attack ?? 0.003,
    Math.max(0.006, opts.duration * 0.72), opts.release ?? Math.max(0.025, opts.duration * 0.28));
  src.connect(filter);
  filter.connect(amp);
  if (opts.drive) {
    const shaper = saturation(ctx);
    amp.connect(shaper);
    shaper.connect(panner);
  } else {
    amp.connect(panner);
  }
  panner.connect(dest);
  src.start(t);
  src.stop(t + opts.duration + (opts.release ?? 0.05) + 0.025);
}

function subImpact(ctx: AudioContext, dest: AudioNode, t: number, weight = 1): void {
  voice(ctx, dest, t, { from: 118, to: 34, duration: 0.22, peak: 0.19 * weight,
    type: 'sine', attack: 0.002, hold: 0.055, release: 0.30 });
  voice(ctx, dest, t + 0.006, { from: 62, to: 30, duration: 0.17,
    peak: 0.07 * weight, type: 'triangle', attack: 0.003, hold: 0.035, release: 0.23, drive: true });
  texture(ctx, dest, t, { duration: 0.18, peak: 0.105 * weight,
    filter: 'lowpass', from: 1050, to: 72, q: 0.7, attack: 0.002, release: 0.20, drive: true });
  // A short upper crack makes the low-frequency hit read clearly on small speakers.
  texture(ctx, dest, t + 0.003, { duration: 0.045, peak: 0.035 * weight,
    filter: 'highpass', from: 6200, to: 1900, q: 0.7, attack: 0.001, release: 0.055, pan: -0.08 });
}
function metalClang(ctx: AudioContext, dest: AudioNode, t: number, weight = 1, pan = 0): void {
  // Slightly inharmonic partials + a bright contact transient + a short plate-like tail.
  voice(ctx, dest, t, { from: 740, to: 505, duration: 0.11, peak: 0.052 * weight,
    type: 'triangle', hold: 0.025, release: 0.27, pan });
  voice(ctx, dest, t + 0.002, { from: 1193, to: 876, duration: 0.09, peak: 0.031 * weight,
    type: 'sine', hold: 0.018, release: 0.21, pan: pan + 0.08 });
  voice(ctx, dest, t + 0.004, { from: 1877, to: 1420, duration: 0.07, peak: 0.015 * weight,
    type: 'sine', hold: 0.012, release: 0.16, pan: pan - 0.08 });
  texture(ctx, dest, t, { duration: 0.055, peak: 0.042 * weight,
    filter: 'highpass', from: 7600, to: 2100, q: 0.75, attack: 0.001, release: 0.075, pan });
}

function crystal(ctx: AudioContext, dest: AudioNode, t: number, root: number, weight = 1, pan = 0): void {
  voice(ctx, dest, t, { from: root * 2.73, to: root * 2.68, duration: 0.16,
    peak: 0.027 * weight, type: 'sine', hold: 0.035, release: 0.42, pan });
  voice(ctx, dest, t + 0.003, { from: root * 4.11, to: root * 3.96, duration: 0.12,
    peak: 0.013 * weight, type: 'sine', hold: 0.018, release: 0.30, pan: pan + 0.1 });
  texture(ctx, dest, t, { duration: 0.045, peak: 0.014 * weight,
    filter: 'highpass', from: 9200, to: 3800, q: 0.6, release: 0.09, pan });
}

function whoosh(ctx: AudioContext, dest: AudioNode, t: number, duration: number,
                peak: number, rising = false, pan = 0): void {
  texture(ctx, dest, t, { duration, peak, filter: 'bandpass',
    from: rising ? 420 : 5200, to: rising ? 5200 : 430,
    q: 0.85, attack: 0.012, release: 0.045, pan });
}

function pulseTrain(ctx: AudioContext, dest: AudioNode, t: number, count: number,
                    interval: number, fn: (at: number, i: number) => void): void {
  for (let i = 0; i < count; i++) fn(t + i * interval, i);
}

// FLOOR 1 — WARDEN MALAKOR: forged iron, plate resonance, brutal mass.
function malakor(ctx: AudioContext, d: AudioNode, a: string, t: number): boolean {
  switch (a) {
    case 'iron_bulwark':
      // Shield locks into place, a forceful clang, then a resonant defensive hum.
      texture(ctx, d, t, { duration: 0.10, peak: 0.045, filter: 'bandpass', from: 2400, to: 620, q: 2.8 });
      metalClang(ctx, d, t + 0.075, 1.2, -0.12);
      voice(ctx, d, t + 0.09, { from: 164, to: 118, duration: 0.24, peak: 0.09, type: 'triangle', release: 0.48, drive: true });
      [0, 1, 2].forEach(i => voice(ctx, d, t + 0.13 + i * 0.012, {
        from: [392, 587, 784][i], to: [386, 574, 765][i], duration: 0.22,
        peak: [0.025, 0.017, 0.012][i], release: 0.62, pan: (i - 1) * 0.18
      }));
      return true;
    case 'triple_cleave':
      pulseTrain(ctx, d, t, 3, 0.17, (at, i) => {
        whoosh(ctx, d, at, 0.12 + i * 0.012, 0.073 - i * 0.012, true, (i - 1) * 0.24);
        voice(ctx, d, at + 0.025, { from: 680 - i * 95, to: 145 - i * 12,
          duration: 0.10, peak: 0.043 - i * 0.006, type: 'sawtooth',
          attack: 0.002, release: 0.075, drive: true, pan: (i - 1) * 0.2 });
        texture(ctx, d, at + 0.018, { duration: 0.07, peak: 0.026,
          filter: 'highpass', from: 5400, to: 1700, q: 0.8, pan: (i - 1) * 0.2 });
        if (i === 2) metalClang(ctx, d, at + 0.045, 0.8, 0.12);
      });
      return true;
    case 'bastion_ground_slam':
      whoosh(ctx, d, t, 0.22, 0.045, false);
      voice(ctx, d, t + 0.14, { from: 96, to: 48, duration: 0.12, peak: 0.055, type: 'sine', release: 0.12 });
      subImpact(ctx, d, t + 0.23, 1.35);
      metalClang(ctx, d, t + 0.245, 0.75);
      texture(ctx, d, t + 0.25, { duration: 0.34, peak: 0.075, filter: 'lowpass',
        from: 1500, to: 95, q: 0.65, attack: 0.004, release: 0.22, drive: true });
      return true;
    case 'war_charge':
      whoosh(ctx, d, t, 0.44, 0.095, true);
      voice(ctx, d, t + 0.015, { from: 68, to: 152, duration: 0.39, peak: 0.075,
        type: 'sawtooth', release: 0.11, drive: true });
      pulseTrain(ctx, d, t + 0.10, 4, 0.09, (at, i) => metalClang(ctx, d, at, 0.25 + i * 0.035, (i % 2 ? 1 : -1) * 0.18));
      subImpact(ctx, d, t + 0.47, 0.78);
      return true;
  }
  return false;
}

        if (i % 2 === 0) crystal(ctx, d, t + i * 0.078 + 0.018, f * 0.72, 0.42, -pan);
      });
      // Final chord resolves the spiral rather than simply stopping the arpeggio.
      [523.25, 659.25, 783.99, 1046.5].forEach((f, i) =>
        voice(ctx, d, t + 0.67, { from: f, to: f * 0.998, duration: 0.25,
          peak: 0.018, type: 'sine', release: 0.78, pan: (i - 1.5) * 0.2 }));
      texture(ctx, d, t + 0.10, { duration: 0.70, peak: 0.022,
        filter: 'highpass', from: 10200, to: 4900, q: 0.65, attack: 0.08, release: 0.30 });
      return true;
    }
  }
  return false;
}

/** Returns true only when a recognized Floor 1–5 boss ability was scheduled. */
export function playBossAbilitySfx(
  ctx: AudioContext,
  dest: AudioNode,
  boss: string,
  ability: string
): boolean {
  if (!ctx || !dest || ctx.state === 'closed') return false;
  const b = boss.toLowerCase().replace(/[^a-z]/g, '');
  const a = ability.toLowerCase().replace(/['’]/g, '').trim()
    .replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '');
  const t = ctx.currentTime + 0.008;
  const id: BossId | null =
    b.includes('malakor') || b === 'warden' ? 'malakor' :
    b.includes('vaelor') || b === 'archlich' ? 'vaelor' :
    b.includes('ignis') ? 'ignis' :
    b.includes('malice') ? 'malice' :
    b.includes('archon') ? 'archon' : null;
  if (!id) return false;
  switch (id) {
    case 'malakor': return malakor(ctx, dest, a, t);
    case 'vaelor': return vaelor(ctx, dest, a, t);
    case 'ignis': return ignis(ctx, dest, a, t);
    case 'malice': return malice(ctx, dest, a, t);
    case 'archon': return archon(ctx, dest, a, t);
  }
}
